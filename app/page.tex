"use client"
import { useState } from "react"
export default function Page(){
  const [img,setImg]=useState<any>(null)
  const [prompt,setPrompt]=useState("")
  const [loading,setLoading]=useState(false)
  const [done,setDone]=useState(false)
  return(
    <div style={{minHeight:'100vh',background:'black',color:'white',display:'flex',flexDirection:'column',alignItems:'center',padding:'16px',fontFamily:'sans-serif'}}>
      <h1 style={{fontSize:'48px',fontWeight:'900',marginTop:'40px'}}>Aitrix <span style={{color:'#a855f7'}}>Pro</span></h1>
      <p style={{color:'#71717a',marginTop:'8px'}}>No Login - Generate Viral Videos</p>
      <div style={{width:'100%',maxWidth:'700px',marginTop:'40px',background:'#18181b',border:'1px solid #27272a',borderRadius:'24px',padding:'24px'}}>
        <label style={{width:'100%',height:'240px',border:'2px dashed #3f3f46',borderRadius:'16px',display:'flex',alignItems:'center',justifyContent:'center',cursor:'pointer',overflow:'hidden'}}>
          <input type="file" hidden onChange={e=>setImg(URL.createObjectURL(e.target.files![0]))}/>
          {img?<img src={img} style={{height:'100%',width:'100%',objectFit:'cover'}}/>:<span>📸 Click to Upload Image</span>}
        </label>
        <input value={prompt} onChange={e=>setPrompt(e.target.value)} placeholder="Describe motion... e.g. camera zoom in, smile" style={{width:'100%',marginTop:'16px',padding:'16px',background:'#27272a',border:'none',borderRadius:'12px',color:'white'}}/>
        <button onClick={()=>{setLoading(true);setTimeout(()=>{setLoading(false);setDone(true)},2500)}} style={{width:'100%',marginTop:'16px',background:'white',color:'black',fontWeight:'bold',padding:'16px',borderRadius:'12px',border:'none',cursor:'pointer',fontSize:'16px'}}>
          {loading?"⏳ Generating Magic...":"🚀 Generate Video"}
        </button>
        {done && <div style={{marginTop:'16px',padding:'16px',background:'#14532d',borderRadius:'12px',textAlign:'center'}}>✅ Video Ready! (Demo Version - Connect API later)</div>}
      </div>
      <p style={{marginTop:'20px',color:'#52525b',fontSize:'12px'}}>Free version - No Google Login needed</p>
    </div>
  )
}
